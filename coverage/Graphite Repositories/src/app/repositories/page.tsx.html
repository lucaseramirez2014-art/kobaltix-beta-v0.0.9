"use client";
import React,{useEffect,useState} from "react";
import Link from "next/link";
import {useRouter} from "next/navigation";
import {Sidebar} from "@/components/Sidebar";
import {Header} from "@/components/Header";
import {RepositoryLibrary} from "@/components/RepositoryLibrary";
import {MOCK_REPOSITORIES} from "@/data/mockData";
import {Repository} from "@/types";
export default function RepositoriesPage(){const router=useRouter();const [open,setOpen]=useState(false),[search,setSearch]=useState(""),[repos,setRepos]=useState<Repository[]>(MOCK_REPOSITORIES),[loading,setLoading]=useState(true),[error,setError]=useState(""),[url,setUrl]=useState(""),[message,setMessage]=useState(""),[importing,setImporting]=useState(false);
 const load=()=>{setLoading(true);fetch("/api/github/repos").then(async r=>{const d=await r.json();if(!r.ok)throw Error(d.error||"Unable to load repositories");setRepos(d.repositories);setError("")}).catch(e=>setError(e.message)).finally(()=>setLoading(false))};useEffect(()=>{const id=window.setTimeout(load,0);return()=>window.clearTimeout(id)},[]);
 async function submit(e:React.FormEvent){e.preventDefault();setMessage("");setError("");setImporting(true);try{const r=await fetch("/api/github/import",{method:"POST",headers:{"Content-Type":"application/json"},body:JSON.stringify({url})});const d=await r.json();if(!r.ok)throw Error(d.error||"Import failed");setUrl("");setMessage(`Imported ${d.owner}/${d.name}`);load()}catch(e){setError(e instanceof Error?e.message:"Import failed")}finally{setImporting(false)}}
 return <div className="min-h-screen bg-[#0c0d0e] text-[#ededed] flex"><Sidebar currentTab="repositories" onSelectTab={t=>{if(t==="dashboard")router.push("/");else if(t==="repositories")router.push("/repositories");else router.push(`/${t}`)}} isOpenMobile={open} onCloseMobile={()=>setOpen(false)}/><div className="flex-1 flex flex-col min-w-0 lg:pl-60"><Header currentTab="repositories" onOpenMobileSidebar={()=>setOpen(true)} searchQuery={search} onSearchChange={setSearch}/><main className="flex-1 p-4 sm:p-6 space-y-4 max-w-6xl w-full mx-auto"><div className="flex items-center justify-between pb-3 border-b border-[#23262d]"><div><h1 className="text-base font-semibold">Repositories</h1><p className="text-xs text-[#71717a] mt-0.5">{loading?"Loading repositories…":error?error:`${repos.length} repositories`} <span className="ml-2 text-[#71717a]">Live and imported sources are labeled below.</span></p></div></div><form onSubmit={submit} className="border border-[#23262d] rounded p-3 flex flex-col sm:flex-row gap-2"><input required value={url} onChange={e=>setUrl(e.target.value)} placeholder="Import GitHub URL or owner/name" className="flex-1 bg-[#121316] border border-[#23262d] rounded px-3 py-2 text-xs"/><button disabled={importing} className="bg-[#4285F4] text-white rounded px-4 py-2 text-xs disabled:opacity-50">{importing?"Importing…":"Import repository"}</button></form>{message&&<p className="text-xs text-[#34A853]">{message}</p>}{error&&<p className="text-xs text-[#EA4335]">{error}</p>}<RepositoryLibrary repositories={repos} initialSearch={search}/></main><footer className="border-t border-[#1c1e24] py-3 px-6 text-xs text-[#52525b] font-mono flex justify-between max-w-6xl w-full mx-auto"><Link href="/">← Dashboard</Link><span>Graphite Repo</span></footer></div></div>}
