import crypto from "node:crypto";
import { NextResponse } from "next/server";
import { scanRepository } from "@/lib/intelligence/repositoryScan";
import { appendActivity, saveScan, scans } from "@/lib/store";
const valid = /^[A-Za-z0-9_.-]{1,100}$/;
export async function GET(req: Request) { const u = new URL(req.url), owner = u.searchParams.get("owner"), name = u.searchParams.get("name"), ref = u.searchParams.get("ref"); if (!owner || !name) return NextResponse.json({ error: "owner and name required" }, { status: 400 }); return NextResponse.json({ scans: (await scans()).filter(s => s.owner === owner && s.repository === name && (!ref || s.ref === ref)) }); }
export async function POST(req: Request) { try { const b = await req.json(), owner = String(b.owner || ""), name = String(b.name || b.repository || ""), ref = String(b.ref || "main"); if (!valid.test(owner) || !valid.test(name) || !ref || ref.length > 200) return NextResponse.json({ error: "valid owner, name, and ref are required" }, { status: 400 }); const result = await saveScan(await scanRepository(owner, name, ref)); await appendActivity({ id: crypto.randomUUID(), type: "security_scan", title: `Repository scan: ${owner}/${name} (${ref})`, repository: name, author: { name: "Graphite", handle: "system" }, timestamp: new Date().toISOString(), statusBadge: { text: result.status === "complete" ? "Scan complete" : "Scan partial", variant: result.status === "complete" ? "green" : "yellow" } }); return NextResponse.json(result); } catch (e) { return NextResponse.json({ error: e instanceof Error ? e.message : "Repository scan failed" }, { status: 502 }); } }
