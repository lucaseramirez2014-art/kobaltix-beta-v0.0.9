import { NextResponse } from "next/server";
import { repository } from "@/lib/github";
import { addImported } from "@/lib/importedRepos";
import { appendActivity } from "@/lib/store";
import crypto from "node:crypto";
export async function POST(request: Request) { const body = await request.json(); let { owner, name } = body; if (body.url) { try { const u = new URL(body.url); if (u.hostname !== "github.com") throw Error(); [owner,name] = u.pathname.split("/").filter(Boolean); } catch { return NextResponse.json({error:"Enter owner/name or a valid github.com URL"},{status:400}); } } if (!/^[A-Za-z0-9_.-]{1,100}$/.test(owner||"") || !/^[A-Za-z0-9_.-]{1,100}$/.test(name||"")) return NextResponse.json({error:"Invalid owner or repository name"},{status:400}); try { const r = await repository(owner, name); const value = await addImported({ id: String(r.id), name: r.name, owner: r.owner.login, description: r.description || "", language: r.language || "Unknown", stars: r.stargazers_count, forks: r.forks_count, openIssues: r.open_issues_count, pullRequests: 0, updatedAt: r.updated_at, visibility: (r.visibility || "public") as "public" | "private" | "internal", healthScore: 0, healthStatus: "healthy", intelligenceEnabled: false, branch: r.default_branch }); await appendActivity({id:crypto.randomUUID(),type:"commit",title:`Repository imported: ${value.owner}/${value.name}`,repository:value.name,author:{name:"Graphite",handle:"system"},timestamp:new Date().toISOString(),statusBadge:{text:"Import",variant:"green"}}); return NextResponse.json(value); } catch { return NextResponse.json({ error: "Unable to import repository" }, { status: 502 }); } }
