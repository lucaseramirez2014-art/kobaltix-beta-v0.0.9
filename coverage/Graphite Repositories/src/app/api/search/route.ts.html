import {NextResponse} from "next/server"; import {repositories,tasks,analyses,architectures} from "@/lib/store";
export async function GET(req:Request){const q=(new URL(req.url).searchParams.get("q")||"").toLowerCase().trim();if(!q)return NextResponse.json({results:[]});const [rs,ts,as,arch]=await Promise.all([repositories(),tasks(),analyses(),architectures()]);const results=[...rs.filter(x=>`${x.owner}/${x.name} ${x.description}`.toLowerCase().includes(q)).map(x=>({type:"repository",title:`${x.owner}/${x.name}`,detail:x.description})),...ts.filter(x=>`${x.title} ${x.description}`.toLowerCase().includes(q)).map(x=>({type:"task",title:x.title,detail:x.status})),...as.filter(x=>`${x.owner}/${x.repository} ${x.base} ${x.head}`.toLowerCase().includes(q)).map(x=>({type:"analysis",title:`${x.owner}/${x.repository}`,detail:`${x.base} → ${x.head}`})),...arch.filter(x=>`${x.owner}/${x.repository} ${x.ref}`.toLowerCase().includes(q)).map(x=>({type:"architecture",title:`${x.owner}/${x.repository}`,detail:x.ref}))];return NextResponse.json({results:results.slice(0,50),source:"stored"})}
