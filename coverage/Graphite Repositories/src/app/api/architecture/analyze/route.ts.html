import {NextResponse} from "next/server";import {buildArchitecture} from "@/lib/intelligence/architecture";import crypto from "node:crypto";import {architectures,saveArchitecture,appendActivity} from "@/lib/store";
export async function GET(req:Request){const u=new URL(req.url),owner=u.searchParams.get("owner"),name=u.searchParams.get("name");if(!owner||!name)return NextResponse.json({error:"owner and name required"},{status:400});return NextResponse.json({analyses:(await architectures()).filter(a=>a.owner===owner&&a.repository===name)});}export async function POST(req:Request){try{const b=await req.json(),owner=String(b.owner||""),name=String(b.name||b.repository||""),ref=String(b.ref||"main");if(!/^[\w.-]{1,100}$/.test(owner)||!/^[\w.-]{1,100}$/.test(name))return NextResponse.json({error:"valid owner and name required"},{status:400});const saved=await saveArchitecture(await buildArchitecture(owner,name,ref));await appendActivity({id:crypto.randomUUID(),type:"commit",title:`Architecture analysis: ${owner}/${name} (${ref})`,repository:name,author:{name:"Graphite",handle:"system"},timestamp:new Date().toISOString(),statusBadge:{text:"Architecture",variant:"blue"}});return NextResponse.json(saved);}catch(e){return NextResponse.json({error:e instanceof Error?e.message:"Unable to analyze architecture"},{status:502});}}
