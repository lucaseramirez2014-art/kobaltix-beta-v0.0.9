import {NextResponse} from "next/server";import crypto from "node:crypto";import {deleteRule,rules,saveRule,appendActivity} from "@/lib/store";export async function GET(req:Request){const u=new URL(req.url),owner=u.searchParams.get("owner"),repository=u.searchParams.get("name");if(!owner||!repository)return NextResponse.json({error:"owner and name required"},{status:400});return NextResponse.json({rules:await rules(owner,repository)});}export async function POST(req:Request){const b=await req.json(),owner=String(b.owner||""),repository=String(b.repository||b.name||""),type=b.type,pattern=String(b.pattern||"");if(!owner||!repository||!["forbidden-import","directory-boundary"].includes(type)||!pattern)return NextResponse.json({error:"owner, repository, valid type and pattern required"},{status:400});const rule={id:crypto.randomUUID(),owner,repository,type,pattern,from:b.from?String(b.from):undefined,to:b.to?String(b.to):undefined,enabled:b.enabled!==false,createdAt:new Date().toISOString()} as const;const saved=await saveRule(rule);await appendActivity({id:crypto.randomUUID(),type:"alert",title:`Architecture rule added: ${pattern}`,repository,author:{name:"Graphite",handle:"system"},timestamp:new Date().toISOString(),statusBadge:{text:"Rule",variant:"blue"}});return NextResponse.json(saved);}export async function DELETE(req:Request){const id=new URL(req.url).searchParams.get("id");if(!id)return NextResponse.json({error:"id required"},{status:400});await deleteRule(id);await appendActivity({id:crypto.randomUUID(),type:"alert",title:`Architecture rule removed: ${id}`,repository:"Architecture",author:{name:"Graphite",handle:"system"},timestamp:new Date().toISOString(),statusBadge:{text:"Rule",variant:"red"}});return NextResponse.json({ok:true});}
