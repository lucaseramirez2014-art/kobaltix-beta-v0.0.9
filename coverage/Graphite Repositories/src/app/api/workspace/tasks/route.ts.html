import crypto from "node:crypto";
import {NextResponse} from "next/server"; import {tasks,saveTask,deleteTask,Task,appendActivity} from "@/lib/store";
const statuses=["backlog","todo","doing","done"] as const, priorities=["low","medium","high"] as const;
function taskFrom(b:Record<string,unknown>,id:string):Task{const title=String(b.title||"").trim();if(!title||title.length>240)throw Error("title is required (max 240 characters)");const status=(b.status||"todo") as Task["status"],priority=(b.priority||"medium") as Task["priority"];if(!statuses.includes(status)||!priorities.includes(priority))throw Error("invalid status or priority");return {id,title,description:String(b.description||""),status,priority,labels:Array.isArray(b.labels)?b.labels.map(String):[],milestone:b.milestone?String(b.milestone):undefined,comments:Array.isArray(b.comments)?b.comments.map(String):[],createdAt:String(b.createdAt||new Date().toISOString()),updatedAt:new Date().toISOString(),repository:b.repository?String(b.repository):undefined,findingId:b.findingId?String(b.findingId):undefined}}
export async function GET(){return NextResponse.json({tasks:await tasks(),source:"stored"})} export async function POST(req:Request){try{const t=taskFrom(await req.json(),crypto.randomUUID());await appendActivity({id:crypto.randomUUID(),type:"alert",title:`Task created: ${t.title}`,repository:t.repository||"Workspace",author:{name:"Graphite",handle:"system"},timestamp:new Date().toISOString(),statusBadge:{text:"Task",variant:"blue"}});return NextResponse.json(await saveTask(t),{status:201})}catch(e){return NextResponse.json({error:e instanceof Error?e.message:"Invalid task"},{status:400})}}
export async function PATCH(req:Request){try{const b=await req.json(),id=String(b.id||"");const old=(await tasks()).find(t=>t.id===id);if(!old)return NextResponse.json({error:"task not found"},{status:404});const updated=taskFrom({...old,...b},id);await appendActivity({id:crypto.randomUUID(),type:"alert",title:`Task updated: ${updated.title}`,repository:updated.repository||"Workspace",author:{name:"Graphite",handle:"system"},timestamp:new Date().toISOString(),statusBadge:{text:"Task",variant:"blue"}});return NextResponse.json(await saveTask(updated))}catch(e){return NextResponse.json({error:e instanceof Error?e.message:"Invalid task"},{status:400})}}
export async function DELETE(req:Request){const id=new URL(req.url).searchParams.get("id");if(!id)return NextResponse.json({error:"id required"},{status:400});const old=(await tasks()).find(t=>t.id===id);await deleteTask(id);if(old)await appendActivity({id:crypto.randomUUID(),type:"alert",title:`Task deleted: ${old.title}`,repository:old.repository||"Workspace",author:{name:"Graphite",handle:"system"},timestamp:new Date().toISOString(),statusBadge:{text:"Task",variant:"red"}});return NextResponse.json({ok:true})}
